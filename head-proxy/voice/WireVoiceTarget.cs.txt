// head-proxy/voice: point the base's empty Head_Proxy source slot at the consumer's VoiceTarget, as a
// property override in a prefab (a configuration prefab or variant). Paste as the body of a static
// editor method returning a string, or into MCP for Unity's execute_code. Idempotent; prints one line:
//   wired | already wired | FAIL: <why>  (a FAIL writes nothing)

const string ROW = "Packages/com.ryan6vrc.patterns/head-proxy/voice/HeadProxyVoiceRig.prefab";
const int SLOT = 1;

if (UnityEditor.AssetDatabase.LoadMainAssetAtPath(ROW) == null) return "FAIL: no prefab at " + ROW;
var root = UnityEditor.PrefabUtility.LoadPrefabContents(ROW);
try {
    UnityEngine.Component constraint = null; int proxies = 0;
    UnityEngine.Transform socket = null; int sockets = 0;
    foreach (var t in root.GetComponentsInChildren<UnityEngine.Transform>(true)) {
        if (t.name == "VoiceTarget") { socket = t; sockets++; }
        if (t.name != "Head_Proxy") continue;
        foreach (var c in t.GetComponents<UnityEngine.Component>())
            if (c != null && c.GetType().Name == "VRCPositionConstraint") { constraint = c; proxies++; }
    }
    if (proxies != 1) return "FAIL: " + proxies + " Head_Proxy VRCPositionConstraints under " + ROW;
    if (sockets != 1) return "FAIL: " + sockets + " objects named VoiceTarget under " + ROW;

    var so = new UnityEditor.SerializedObject(constraint);
    int len = so.FindProperty("Sources.totalLength").intValue;
    if (len <= SLOT) return "FAIL: Head_Proxy has " + len + " source(s); the base must serialize slot " + SLOT;
    var slot = so.FindProperty("Sources.source" + SLOT + ".SourceTransform");
    var current = slot.objectReferenceValue as UnityEngine.Transform;
    if (current == socket) return "already wired";
    if (current != null) return "FAIL: slot " + SLOT + " names " + current.name + ", not VoiceTarget";

    slot.objectReferenceValue = socket;
    so.ApplyModifiedPropertiesWithoutUndo();
    UnityEditor.PrefabUtility.SaveAsPrefabAsset(root, ROW);
    return "wired";
} finally {
    UnityEditor.PrefabUtility.UnloadPrefabContents(root);
}
